import React, { useState } from 'react';
import * as LucideIcons from 'lucide-react';
import { Ruler } from 'lucide-react';
const { 
  ImageIcon, FileText, CheckCircle, Lock, Building2, User, Users, ChevronDown, ChevronUp 
} = LucideIcons;
import { StatusBadge } from '../ui/StatusBadge';
import { consolidateCommercialItems } from '../../utils/orderItemConsolidation';

export interface ProductionViewProps {
  orders: any[];
  setViewingLogosOrderId: (id: string | null) => void;
  setViewingInvoiceId: (id: string | null) => void;
  isWriteDisabled: (view: string) => boolean;
  updateStatus: (id: string, status: string) => Promise<void>;
  setExpandedLogoUrl: (url: string | null) => void;
  onOpenEditRoster?: (order: any, readOnly?: boolean) => void;
}

export const ProductionView: React.FC<ProductionViewProps> = ({
  orders,
  setViewingLogosOrderId,
  setViewingInvoiceId,
  isWriteDisabled,
  updateStatus,
  setExpandedLogoUrl,
  onOpenEditRoster
}) => {
  const [expandedRosterOrderId, setExpandedRosterOrderId] = useState<string | null>(null);
  const [expandedMeasurements, setExpandedMeasurements] = useState<Record<string, boolean>>({});

  const toggleMeasurements = (idx: string) => {
    setExpandedMeasurements(prev => ({ ...prev, [idx]: !prev[idx] }));
  };

  return (
    <div className="w-full space-y-6 animate-in fade-in">
      <div className="grid grid-cols-1 lg:grid-cols-2 xl:grid-cols-3 gap-4 w-full">
        {orders.filter(o => ['Material Listo', 'En Producción', 'En Confección'].includes(o.status)).map(order => {
          const isInstitutional = Boolean(order.isInstitutional || (order.institutionalRoster && order.institutionalRoster.length > 0) || (order.items || []).some((it: any) => Boolean(it.recipientName)));
          const groupedItems = consolidateCommercialItems(order.items || [], isInstitutional);
          const isRosterExpanded = expandedRosterOrderId === order.id;

          return (
            <div key={order.id} className="bg-stone-900 border border-stone-800 p-3.5 sm:p-4 rounded-2xl shadow-sm flex flex-col justify-between gap-3 relative w-full overflow-hidden">
              <div className="flex flex-col sm:flex-row sm:justify-between items-start sm:items-center gap-2.5 sm:gap-2 select-none w-full">
                <div className="flex flex-wrap items-center gap-2 min-w-0 w-full sm:w-auto">
                  <h3 className="text-lg sm:text-xl font-black text-stone-200 uppercase tracking-tight truncate max-w-full sm:max-w-[180px] xl:max-w-[150px] 2xl:max-w-[200px]">{order.customer}</h3>
                  {isInstitutional && (
                    <span className="bg-amber-500/10 text-amber-400 border border-amber-500/20 text-[9px] font-black px-1.5 py-0.5 rounded-md uppercase tracking-wider flex items-center gap-1 shrink-0">
                      <Building2 size={10} /> Inst. ({order.items?.length || 0})
                    </span>
                  )}
                  <div className="shrink-0 scale-90 origin-left">
                    <StatusBadge status={order.status} />
                  </div>
                </div>
                <div className="flex items-center gap-1.5 w-full sm:w-auto overflow-x-auto custom-scrollbar pb-1 sm:pb-0 shrink-0">
                  {isInstitutional && (
                    <button 
                      onClick={() => setExpandedRosterOrderId(isRosterExpanded ? null : order.id)}
                      className="bg-amber-950/40 hover:bg-amber-900/60 text-amber-300 border border-amber-800/60 py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider flex items-center justify-center h-8 shrink-0 transition cursor-pointer"
                    >
                      <Users size={12} /> {isRosterExpanded ? 'Ocultar' : `Nómina (${order.items?.length || 0})`}
                    </button>
                  )}
                  {order.logos && order.logos.length > 0 && (
                    <button onClick={() => setViewingLogosOrderId(order.id)} className="bg-stone-800 hover:bg-stone-700 text-amber-500 border border-stone-700 py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider flex items-center justify-center h-8 shrink-0 transition">
                      <ImageIcon size={12}/> Logos ({order.logos.length})
                    </button>
                  )}
                  <button onClick={() => setViewingInvoiceId(order.id)} className="bg-stone-800 hover:bg-stone-700 text-stone-300 border border-stone-700 py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider flex items-center justify-center h-8 shrink-0 transition">
                    <FileText size={12}/> Orden
                  </button>
                  {order.status === 'Material Listo' ? (
                    !isWriteDisabled('produccion') ? (
                      <button onClick={() => updateStatus(order.id, 'En Producción')} className="bg-purple-600 text-white py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider hover:bg-purple-500 transition shadow-lg cursor-pointer flex items-center justify-center gap-1 h-8 shrink-0">Iniciar</button>
                    ) : (
                      <span className="bg-stone-850 border border-stone-800 text-stone-500 py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider select-none flex items-center justify-center gap-1 cursor-not-allowed h-8 shrink-0">
                        Solo Lectura <Lock size={10}/>
                      </span>
                    )
                  ) : (
                    !isWriteDisabled('produccion') ? (
                      <button onClick={() => updateStatus(order.id, 'Producido')} className="bg-indigo-600 text-white py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider hover:bg-indigo-500 transition shadow-lg flex items-center justify-center gap-1 cursor-pointer h-8 shrink-0"><CheckCircle size={12}/> Terminado</button>
                    ) : (
                      <span className="bg-stone-850 border border-stone-800 text-stone-500 py-1.5 px-3 rounded-lg font-black uppercase text-[10px] tracking-wider select-none flex items-center justify-center gap-1 cursor-not-allowed h-8 shrink-0">
                        Solo Lectura <Lock size={10}/>
                      </span>
                    )
                  )}
                </div>
              </div>
              {/* Grouped Production Garments Summary */}
              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                {groupedItems.map((item: any, idx: number) => {
                  
                  // 1. Resolve Filipina Specs (8 Points)
                  const filSibling = (order.items || []).find((it: any) => it.name?.toLowerCase().includes('filipina'));
                  const filMeasurements = (filSibling?.measurements && Object.keys(filSibling.measurements).length > 0)
                    ? filSibling.measurements
                    : (item.name?.toLowerCase().includes('filipina') && item.measurements ? item.measurements : (order.measurements?.filipina || order.measurements || order.contact?.measurements?.filipina || order.contact?.measurements || null));

                  // 2. Resolve Pantalón Specs (4 Points)
                  const panSibling = (order.items || []).find((it: any) => it.name?.toLowerCase().includes('pantalon') || it.name?.toLowerCase().includes('pantalón'));
                  const panMeasurements = (panSibling?.measurements && Object.keys(panSibling.measurements).length > 0)
                    ? panSibling.measurements
                    : ((item.name?.toLowerCase().includes('pantalon') || item.name?.toLowerCase().includes('pantalón')) && item.measurements ? item.measurements : (order.measurements?.pantalon || order.measurements || order.contact?.measurements?.pantalon || order.contact?.measurements || null));

                  // 3. Helper to verify if measurements have real values
                  const hasValues = (obj: any) => obj && Object.values(obj).some((v: any) => v !== undefined && v !== null && String(v).trim() !== '' && String(v).trim() !== '0');

                  const hasFilSpecs = hasValues(filMeasurements);
                  const hasPanSpecs = hasValues(panMeasurements);
                  const hasDirectSpecs = hasValues(item.measurements);
                  
                  // Specific overrides for bundled uniforms
                  const isBundle = item.name?.toLowerCase().includes('completo udi');
                  const isFilipina = item.name?.toLowerCase().includes('filipina');
                  const isPantalon = item.name?.toLowerCase().includes('pantal');
                  
                  const showFil = hasFilSpecs && (isBundle || isFilipina);
                  const showPan = hasPanSpecs && (isBundle || isPantalon);
                  
                  const isCustom = showFil || showPan || (hasDirectSpecs && !isFilipina && !isPantalon && !isBundle);

                  return (
                  <div key={idx} className="bg-stone-950 p-4 rounded-2xl border border-stone-800 flex flex-col justify-between">
                    <div>
                      {item.referenceImages && item.referenceImages.length > 0 && (
                        <div className="mb-3 bg-stone-900/40 p-2 rounded-xl border border-stone-850">
                          <span className="text-[9px] font-black text-stone-500 uppercase tracking-widest block mb-1.5">Imágenes de Referencia</span>
                          <div className="flex flex-wrap gap-1.5">
                            {item.referenceImages.map((img: any, imgIdx: number) => (
                              <div key={imgIdx} className="relative group bg-stone-950 border border-stone-850 p-1 rounded-lg flex items-center gap-1">
                                <img 
                                  src={img.url} 
                                  alt="Ref" 
                                  className="w-8 h-8 object-cover rounded border border-stone-800 cursor-zoom-in" 
                                  onClick={() => setExpandedLogoUrl(img.url)} 
                                />
                                {img.note && <span className="text-[8px] font-bold text-stone-400 max-w-[80px] truncate pr-1" title={img.note}>{img.note}</span>}
                              </div>
                            ))}
                          </div>
                        </div>
                      )}

                      <div className="flex justify-between items-start mb-3 pb-2 border-b border-stone-900">
                        <div>
                          <span className="font-black text-stone-200 text-base uppercase tracking-tight block">{item.name}</span>
                          <div className="flex flex-wrap items-center gap-1.5 mt-1">
                            <span className="text-[10px] font-black uppercase text-amber-500 tracking-wider bg-stone-900 border border-stone-800 px-2 py-0.5 rounded font-sans">
                              {item.fabric || 'Sin tela'}{item.fabricColor ? ` • ${item.fabricColor}` : ''}
                            </span>
                            
                                  {item.size && !isCustom && (
                                    <span className="text-[10px] font-black uppercase text-emerald-400 bg-emerald-950/40 border border-emerald-800/60 px-2 py-0.5 rounded font-mono">
                                      Talla: {item.size}
                                    </span>
                                  )}
                                  {isCustom && (
                                    <span className="text-[10px] font-black uppercase text-stone-900 bg-amber-400 border border-amber-500 px-2 py-0.5 rounded font-mono flex items-center gap-1">
                                      <Ruler size={10} /> A MEDIDA
                                    </span>
                                  )}
        
                          </div>
                        </div>
                        <span className="text-amber-400 font-mono font-black text-lg">x{item.qty || 1}</span>
                      </div>

                      
                      
                      {isCustom && (
                        <div className="mb-3">
                          <button 
                            onClick={() => toggleMeasurements(`${order.id}-${idx}`)}
                            className="w-full flex items-center justify-between bg-stone-900 hover:bg-stone-800 border border-stone-800 text-amber-500 px-3 py-2 rounded-xl text-[10px] font-black uppercase tracking-wider transition"
                          >
                            <span className="flex items-center gap-2"><Ruler size={12}/> Medidas de Taller</span>
                            {expandedMeasurements[`${order.id}-${idx}`] ? <ChevronUp size={14}/> : <ChevronDown size={14}/>}
                          </button>
                          
                          {expandedMeasurements[`${order.id}-${idx}`] && (
                            <div className="bg-stone-950/90 border border-amber-500/30 p-2.5 sm:p-3 rounded-xl mt-2 font-mono w-full max-w-full overflow-hidden space-y-3">
                              {showFil && (
                                <div>
                                  <span className="text-[10px] font-black text-amber-500 mb-1.5 block">👘 FILIPINA</span>
                                  <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5 w-full">
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Hombros</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.hombros || filMeasurements.F_Hombros || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Cuello</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.cuello || filMeasurements.F_Cuello || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Pecho</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.anchoTorso || filMeasurements.F_Ancho_Torso || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Largo Torso</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.largoTorso || filMeasurements.F_Largo_Torso || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Cintura</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.cintura || filMeasurements.F_Cintura || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Cadera</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.cadera || filMeasurements.F_Cadera || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Largo Brazo</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.largoBrazo || filMeasurements.F_Largo_Brazo || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Ancho Brazo</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{filMeasurements.anchoBrazo || filMeasurements.F_Ancho_Brazo || '—'} cm</span></div>
                                  </div>
                                </div>
                              )}
                              
                              {showPan && (
                                <div className={showFil ? "pt-2 border-t border-stone-800" : ""}>
                                  <span className="text-[10px] font-black text-amber-500 mb-1.5 block">👖 PANTALÓN</span>
                                  <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5 w-full">
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Cintura</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{panMeasurements.cintura || panMeasurements.P_Cintura || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Cadera</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{panMeasurements.cadera || panMeasurements.P_Cadera || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Ancho Pierna</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{panMeasurements.anchoPierna || panMeasurements.P_Ancho_Pierna || '—'} cm</span></div>
                                    <div className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">Largo Pierna</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{panMeasurements.largoPierna || panMeasurements.P_Largo_Pierna || '—'} cm</span></div>
                                  </div>
                                </div>
                              )}
                              
                              {(!showFil && !showPan && hasDirectSpecs) && (
                                <div>
                                  <span className="text-[10px] font-black text-amber-500 mb-1.5 block">MEDIDAS</span>
                                  <div className="grid grid-cols-2 sm:grid-cols-4 gap-1.5 w-full">
                                    {Object.entries(item.measurements).filter(([_, v]) => v && String(v).trim() !== '' && String(v).trim() !== '0').map(([k, v]) => (
                                      <div key={k} className="bg-stone-900/90 border border-stone-800/80 rounded-md p-1 sm:p-1.5 flex flex-col items-center justify-center text-center w-full"><span className="text-[9px] sm:text-[10px] font-bold text-stone-400 uppercase tracking-wider block truncate w-full">{k}</span><span className="text-xs font-mono font-black text-amber-400 whitespace-nowrap mt-0.5">{String(v)} cm</span></div>
                                    ))}
                                  </div>
                                </div>
                              )}
                            </div>
                          )}
                        </div>
                      )}
        
{item.embroideries && item.embroideries.length > 0 && (
                        <div className="bg-stone-900/60 p-2.5 rounded-xl border border-stone-800 space-y-1 select-none text-left">
                          <div className="text-[9px] font-black text-stone-400 uppercase tracking-widest mb-1">
                            Bordados Requeridos
                          </div>
                          <div className="flex flex-col gap-1">
                            {item.embroideries.map((emb: string, eIdx: number) => {
                              const isBolsillo = emb.toLowerCase().includes('bolsillo');
                              return (
                                <div key={eIdx} className={`text-[11px] font-bold leading-tight flex items-center gap-1.5 ${isBolsillo ? 'text-amber-400 font-black' : 'text-stone-300'}`}>
                                  <span className={`w-1.5 h-1.5 rounded-full shrink-0 ${isBolsillo ? 'bg-amber-400' : 'bg-stone-600'}`}></span>
                                  <span>{emb.startsWith('Otro:') ? emb.split('|')[0].replace('Otro:', '').trim() : emb}</span>
                                </div>
                              );
                            })}
                          </div>
                        </div>
                      )}
                    </div>
                  </div>
                );
              })}
              </div>

              {/* Detailed Roster Drawer if Expanded */}
              {isInstitutional && isRosterExpanded && (
                <div className="bg-stone-950 border border-stone-800 p-4 rounded-2xl space-y-3 animate-in fade-in">
                  <div className="flex items-center justify-between border-b border-stone-800 pb-2">
                    <span className="text-xs font-black uppercase tracking-wider text-amber-400 flex items-center gap-2">
                      <Users size={14} /> Detalle de Nómina Individual ({order.items?.length || 0} prendas asignadas)
                    </span>
                    {onOpenEditRoster && (
                      <button
                        onClick={() => onOpenEditRoster(order, true)}
                        className="px-3 py-1 bg-stone-900 hover:bg-stone-800 text-stone-200 border border-stone-700 rounded-lg text-[10px] font-black uppercase tracking-wider transition cursor-pointer"
                      >
                        VER NÓMINA COMPLETA
                      </button>
                    )}
                  </div>
                  <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-2.5 max-h-80 overflow-y-auto custom-scrollbar p-1">
                    {order.items.map((item: any, itIdx: number) => (
                      <div key={itIdx} className="bg-stone-900/80 border border-stone-800/80 p-2.5 rounded-xl text-xs space-y-1">
                        <div className="flex justify-between items-start">
                          <span className="font-bold text-stone-200 truncate">{item.recipientName || `Prenda #${itIdx + 1}`}</span>
                          <span className="text-[10px] font-mono font-bold text-amber-400 bg-stone-950 px-1.5 py-0.5 rounded">{item.size || 'STD'}</span>
                        </div>
                        <p className="text-[10px] text-stone-400 truncate">{item.name} • {item.fabric || 'Drill'}</p>
                        {item.recipientNotes && (
                          <p className="text-[9px] text-amber-300/80 italic truncate">Obs: {item.recipientNotes}</p>
                        )}
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {order.notes && <div className="text-sm font-bold text-amber-500 bg-amber-950/20 p-4 rounded-xl italic">"{order.notes}"</div>}
            </div>
          );
        })}
      </div>
    </div>
  );
};
