            await setDoc(doc(db, 'masterContacts', docId), {
              accountId: targetAccount,
              role: targetRole,
              status: c.status || 'ACTIVE',
              createdAt: c.createdAt || Date.now()
            }, { merge: true });
          }
        }
      }

      // Step C: Repair Orders
      setAuditStatusMsg('Reparando claves foráneas en orders...');
      for (const o of orders) {
        const missingAccount = !o.accountId || o.accountId === '';
        const missingContact = !o.contactId || o.contactId === '';

        if (missingAccount || missingContact) {
          const updates: any = {};
          if (missingAccount) {
            const fullText = `${o.customer || ''} ${o.companyName || ''} ${o.category || ''}`.toUpperCase();
            if (fullText.includes('GREEN HOUSE')) {
              updates.accountId = 'ACC-002';
            } else {
              updates.accountId = 'ACC-001';
            }
          }
          if (missingContact) {
            updates.contactId = 'CON-001';
          }
          if (!o.createdAt) updates.createdAt = Date.now();
          if (!o.status) updates.status = 'Estimado';

          await updateDoc(doc(db, 'orders', o.id), updates);
        }
      }

      // Step D: Clean or Re-link orphaned orderItems (T4) & orderModifiers (T5)
      setAuditStatusMsg('Limpiando o re-vinculando ítems huérfanos (orderItems & orderModifiers)...');
      try {
        const itemsSnap = await getDocs(collection(db, 'orderItems'));
        const validOrderIds = new Set(orders.map(o => o.id));
        for (const itemDoc of itemsSnap.docs) {
          const item = itemDoc.data();
          if (!item.orderId || !validOrderIds.has(item.orderId)) {
            const convertedOrderId = (item.orderId || '').replace('COT-', 'ORD-');
            if (validOrderIds.has(convertedOrderId)) {
              await updateDoc(doc(db, 'orderItems', itemDoc.id), { orderId: convertedOrderId });
            } else {
              await deleteDoc(itemDoc.ref);
            }
          }
        }
      } catch (e) { console.warn('Error purging orderItems:', e); }

      try {
        const modsSnap = await getDocs(collection(db, 'orderModifiers'));
        const validOrderIds = new Set(orders.map(o => o.id));
        for (const modDoc of modsSnap.docs) {
          const mod = modDoc.data();
          if (!mod.orderId || !validOrderIds.has(mod.orderId)) {
            const convertedOrderId = (mod.orderId || '').replace('COT-', 'ORD-');
            if (validOrderIds.has(convertedOrderId)) {
              await updateDoc(doc(db, 'orderModifiers', modDoc.id), { orderId: convertedOrderId });
            } else {
              await deleteDoc(modDoc.ref);
            }
          }
        }
      } catch (e) { console.warn('Error purging orderModifiers:', e); }

      setAuditStatusMsg('Reparación completada. Re-ejecutando auditoría...');
      await handleRunAudit();
    } catch (err: any) {
      console.error('Error repairing relational integrity:', err);
      setAuditStatusMsg('Error durante la reparación: ' + err.message);
    } finally {
      setIsRepairing(false);
    }
  };
