            fullName: c.fullName || `${c.contactNombre || ''} ${c.contactApellido || ''}`.trim() || 'Sin Nombre',

        const missingContact = !o.contactId || o.contactId === '';
        if (missingAccount || missingContact) {
          ordersMissingKeys.push({
            id: o.id,
            customer: o.customer || o.companyName || 'Sin Cliente',
            docType: o.docType || 'ORDEN',
            reason: missingAccount && missingContact ? 'Sin accountId y Sin contactId' : (missingAccount ? 'Sin accountId' : 'Sin contactId')
          });
        }
      });

      // 3. Audit orderItems
      setAuditStatusMsg('Auditando colección orderItems (T4)...');
      let scannedItemsCount = 0;
      const orphanedOrderItems: any[] = [];
      try {
        const itemsSnap = await getDocs(collection(db, 'orderItems'));
        scannedItemsCount = itemsSnap.size;
        const validOrderIds = new Set(orders.map(o => o.id));
        itemsSnap.docs.forEach(docSnap => {
          const item = docSnap.data();
          if (!item.orderId || !validOrderIds.has(item.orderId)) {
            orphanedOrderItems.push({
              id: docSnap.id,
              orderId: item.orderId || 'N/A',
              name: item.name || item.description || 'Ítem'
            });
          }
        });

              orderId: mod.orderId || 'N/A',
              type: mod.type || 'Modificador'

        console.warn('Error querying orderModifiers for audit:', err);

  const executeHistoricalRoleRepair = async () => {
    try {
      const linksRef = collection(db, 'account_contacts');
      const linksSnap = await getDocs(linksRef);
      const batch = writeBatch(db);
      let changeCount = 0;
      let repairedCount = 0;
      let batchCount = 0;

      for (const docSnap of linksSnap.docs) {
        const data = docSnap.data();
        const linkId = docSnap.id;
        const cId = data.contactId;
        if (!cId) continue;

        const mc = masterContacts.find(m => m.contactId === cId);
        const contactOrders = orders.filter(o => o.contactId === cId || (mc && mc.fullName && o.customer?.toLowerCase() === mc.fullName.toLowerCase()));
        
        // 1. ACCOUNT RECOVERY LOGIC
        let targetAccountId = 'ACC-000';
        
        // Ensure mc.accountId is actually a valid living account
        const isValidMcAccount = mc && mc.accountId && mc.accountId !== 'ACC-000' && mc.accountId !== 'Ninguna' && mc.accountId !== 'null' && mc.accountId.trim() !== '' && clientAccounts.some(acc => acc.accountId === mc.accountId);
        
        if (isValidMcAccount) {
          targetAccountId = mc.accountId;
        } else {
          
          // Check if there is an order with a valid accountId
          const validAccountOrders = contactOrders.filter(o => o.accountId && o.accountId !== 'ACC-000' && o.accountId !== 'Ninguna' && o.accountId !== 'null' && clientAccounts.some(acc => acc.accountId === o.accountId));
          if (validAccountOrders.length > 0) {
            validAccountOrders.sort((a, b) => (Number(b.createdAt) || 0) - (Number(a.createdAt) || 0));
            targetAccountId = validAccountOrders[0].accountId;
          } else {
            // Check companyName or parentAccountName matching an existing clientAccount
            let matchedAcc = null;
            for (const o of contactOrders) {
              const compToMatch = o.parentAccountName || o.companyName;
              if (compToMatch && compToMatch !== 'Ninguna') {
                matchedAcc = clientAccounts.find(acc => acc.accountName?.toLowerCase() === compToMatch.toLowerCase() || (acc.accountName && compToMatch.toLowerCase().includes(acc.accountName.toLowerCase())));
                if (matchedAcc) break;
              }
            }
            if (matchedAcc) {
              targetAccountId = matchedAcc.accountId;
            } else {
              const hasUdiOrder = contactOrders.some(o => o.category === 'UDI' || o.clientType === 'Universidad UDI' || o.clientType === 'Estudiante UDI');
              const hasGhOrder = contactOrders.some(o => o.companyName?.toUpperCase().includes('GREEN HOUSE') || o.parentAccountName?.toUpperCase().includes('GREEN HOUSE'));
              
              if (hasGhOrder) {
                targetAccountId = 'ACC-002';
              } else if (hasUdiOrder) {
                targetAccountId = 'ACC-001';
              } else {
                targetAccountId = 'ACC-000';
              }
            }
          }
        }

        // 2. ROLE ASSIGNMENT LOGIC
        let targetRole = 'OTRO';
        
        if (typeof contactOrders !== 'undefined' && contactOrders.length > 0) {
          const sortedOrders = [...contactOrders].sort((a, b) => (Number(b.createdAt) || 0) - (Number(a.createdAt) || 0));
          if (sortedOrders[0].contactRole) {
            targetRole = sortedOrders[0].contactRole;
          } else if (typeof mc !== 'undefined' && mc.role) {
            targetRole = mc.role;
          }
        } else if (typeof mc !== 'undefined' && mc.role) {
          targetRole = mc.role;
        }

        // Fix invalid states from previous bug
        if (targetAccountId === 'ACC-000') {
          targetRole = 'PARTICULAR';
        } else if (targetAccountId === 'ACC-001') {
          if (!targetRole || targetRole === 'OTRO' || targetRole === 'PARTICULAR') targetRole = 'ESTUDIANTE';
        } else {
          if (!targetRole || targetRole === 'PARTICULAR') targetRole = 'OTRO';
        }

        // Auto-fix masterContact
        if (typeof mc !== 'undefined' && mc) {
          if (mc.accountId !== targetAccountId || mc.role !== targetRole) {
              batch.update(doc(db, 'masterContacts', mc.id || mc.contactId), {
                  accountId: targetAccountId,
                  role: targetRole
              });
              
              if (typeof repairedCount !== 'undefined') repairedCount++;
              batchCount++;
          }
        }

        if (data.accountId !== targetAccountId || data.role !== targetRole) {
          batch.update(doc(db, 'account_contacts', linkId), {
            accountId: targetAccountId,
            role: targetRole
          });
          changeCount++;
          batchCount++;
          if (batchCount >= 400) {
            await batch.commit();
            batchCount = 0;
          }
        }
      }

      if (batchCount > 0) {
        await batch.commit();
      }

      return changeCount;
    } catch (e) {
      console.error('Error in executeHistoricalRoleRepair:', e);
      return 0;
    }

            for (const o of contactOrders) {
              const compToMatch = o.parentAccountName || o.companyName;
              if (compToMatch && compToMatch !== 'Ninguna') {
                matchedAcc = clientAccounts.find(acc => acc.accountName?.toLowerCase() === compToMatch.toLowerCase() || (acc.accountName && compToMatch.toLowerCase().includes(acc.accountName.toLowerCase())));
                if (matchedAcc) break;
              }
            }
            if (matchedAcc) {
              targetAccountId = matchedAcc.accountId;
            } else {
              const hasUdiOrder = contactOrders.some(o => o.category === 'UDI' || o.clientType === 'Universidad UDI' || o.clientType === 'Estudiante UDI');
              const hasGhOrder = contactOrders.some(o => o.companyName?.toUpperCase().includes('GREEN HOUSE') || o.parentAccountName?.toUpperCase().includes('GREEN HOUSE'));
              
              if (hasGhOrder) {
                targetAccountId = 'ACC-002';
              } else if (hasUdiOrder) {
                targetAccountId = 'ACC-001';
              } else {
                targetAccountId = 'ACC-000';
              }
            }
          }
        }

        // Auto-fix masterContact if it had ACC-000 or empty and we found a real account, OR if role differs
        // We will do this AFTER we compute targetRole below!

        // 2. ROLE ASSIGNMENT LOGIC
        let targetRole = 'OTRO';
        
        if (typeof contactOrders !== 'undefined' && contactOrders.length > 0) {
          const sortedOrders = [...contactOrders].sort((a, b) => (Number(b.createdAt) || 0) - (Number(a.createdAt) || 0));
          if (sortedOrders[0].contactRole) {
            targetRole = sortedOrders[0].contactRole;
          } else if (typeof mc !== 'undefined' && mc.role) {
            targetRole = mc.role;
          }
        } else if (typeof mc !== 'undefined' && mc.role) {
          targetRole = mc.role;
        }

        // Fix invalid states from previous bug
        if (targetAccountId === 'ACC-000') {
          targetRole = 'PARTICULAR';
        } else if (targetAccountId === 'ACC-001') {
          if (!targetRole || targetRole === 'OTRO' || targetRole === 'PARTICULAR') targetRole = 'ESTUDIANTE';
        } else {
          if (!targetRole || targetRole === 'PARTICULAR') targetRole = 'OTRO';
        }

        // 3. FIND EXISTING LINKS FOR THIS CONTACT
        const existingLinks = linksSnap.docs.filter(d => d.data().contactId === cId);

        if (existingLinks.length > 0) {
          // Update existing links to the correct account and role
          for (const linkDoc of existingLinks) {
            const data = linkDoc.data();
            if (data.accountId !== targetAccountId || data.role !== targetRole) {
              batch.update(doc(db, 'account_contacts', linkDoc.id), {
                accountId: targetAccountId,
                role: targetRole
              });
              repairedCount++;
              batchCount++;
              if (batchCount >= 400) {
                await batch.commit();
                batchCount = 0;
              }
            }
          }
        } else {
          // Create a new link document if none exists
          currentSeq++;
          const newLinkId = `LNK-${String(currentSeq).padStart(5, '0')}`;
          batch.set(doc(linksRef, newLinkId), {
            linkId: newLinkId,
            contactId: cId,
            accountId: targetAccountId,
            role: targetRole,
            isPrimary: true,
            createdAt: Date.now()
          });
          createdCount++;
          batchCount++;
          if (batchCount >= 400) {
            await batch.commit();
            batchCount = 0;
          }
        }
      }

      if (batchCount > 0) {
         await batch.commit();
      }

      await setDoc(counterRef, { currentSequence: currentSeq }, { merge: true });

      setAuditStatusMsg(`¡Éxito! Enlaces restaurados: ${repairedCount} actualizados, ${createdCount} creados.`);
      setTimeout(() => { setIsAuditing(false); setShowAuditModal(false); }, 3000);

    } catch (err) {
      console.error(err);
      setAuditStatusMsg('Error en la reparación de enlaces.');
      setTimeout(() => setIsAuditing(false), 3000);
    }
  };
